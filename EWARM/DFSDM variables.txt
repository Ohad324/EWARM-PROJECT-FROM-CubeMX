;   250K / 256K → MCKDIV=0 → undefined
;   Fix: SAI_AUDIO_FREQUENCY_16K → MCKDIV=11 (valid, but 11.7% clock mismatch)

; [Bug 4] CKOUTDIV write silently dropped — RM0399 ordering violation
;   Code set DFSDMEN=1 first, then wrote CKOUTDIV
;   RM0399 §30.4.2: CKOUTDIV writable ONLY when DFSDMEN=0
;   Result: Ch0_CHCFGR1=0x80000000 (CKOUTDIV stayed 0 → CKOUT disabled → filter dead)
;   Fix: write CKOUTDIV first, then DFSDMEN=1

; [Bug 5] HAL MCKDIV rounding: floor(49142857/4096000) = 11, not 12
;   MCKDIV=11 → CK1=2.234 MHz → 11.7% faster than DFSDM CKOUT=2.0 MHz
;   Fix: force MCKDIV=12 via direct CR1 write after HAL_SAI_Init
;   MCKDIV=12 → CK1=2.048 MHz → only 2.4% from CKOUT (acceptable for voice)

; [Bug 6] RCC dual-gate not set — CM7 register writes silently no-op
;   __HAL_RCC_DFSDM1_CLK_ENABLE() only sets RCC->APB2ENR (shared/DMA bus clock)
;   On STM32H747 dual-core, CM7 also needs RCC_C1->APB2ENR bit 30 = 1
;   Without C1 gate: DMA can reach DFSDM (shared bus), but CM7 register writes
;   (CKOUTDIV, CHCFGR1, FLTCR1 etc.) may be silently discarded
;   Fix: add __HAL_RCC_C1_DFSDM1_CLK_ENABLE() immediately after the shared enable

; [Bug 7] Wrong DFSDM channel — CH3 instead of CH1 (root cause of all-1s saturation)
;   SAI4 MicPair1 D1 (LR=GND mic on PC1) bridges to DFSDM1_Channel1 (falling edge)
;   CH3 connects to SAI4 MicPair2 D1 → NOT on this board

;   After fix: g_DfsdmBuf[0] = 0xFF119501 = -61035 (all-0s PDM — mic silent or not driving)
;   Status: Bug 7 FIXED — CH1 is correctly bridged to PC1 via SAI4 silicon path

; 11. NVIC HARDWARE INTERRUPT PRIORITIES
; ══════════════════════════════════════════════════════════════════════════════
; Rule: lower number = higher urgency (ARM Cortex-M convention, opposite of FreeRTOS tasks)
; configMAX_SYSCALL_INTERRUPT_PRIORITY = 5
;   → Interrupts at priority 0-4 are "zero-latency" — CANNOT call any FreeRTOS API
;   → Interrupts at priority 5-15 are "FreeRTOS-safe" — CAN call xXxxFromISR() APIs
;
; ┌─────────────────────────────────┬──────────┬────────────┬───────────────────────────────────┐
; │ Interrupt                       │ Priority │ Zone       │ Notes / Source                    │
; ├─────────────────────────────────┼──────────┼────────────┼───────────────────────────────────┤
; │ (Zone A boundary — pure HW)     │  0 – 4   │ ZERO-LAT   │ Cannot call RTOS API              │
; ├─────────────────────────────────┼──────────┼────────────┼───────────────────────────────────┤
; │ DMA1_Stream1  (DFSDM audio DMA) │    5     │ FENCE/SAFE │ voice_recorder.c:1012             │
; │ MDMA          (display MDMA)    │    5     │ FENCE/SAFE │ main.c MX_MDMA_Init               │
; │ EXTI15_10     (blue button PC13)│    5     │ FENCE/SAFE │ voice_recorder.c:965              │
; │ SDMMC1        (SD card)         │    5     │ FENCE/SAFE │ audio_sd.c:650                    │
; │ ─── configMAX_SYSCALL fence ─── │    5     │ ─ FENCE ─  │ lowest priority above fence       │
; ├─────────────────────────────────┼──────────┼────────────┼───────────────────────────────────┤
; │ JPEG          (HW JPEG decoder) │    6     │ SAFE       │ stm32h7xx_hal_msp.c (was 5→fixed) │
; ├─────────────────────────────────┼──────────┼────────────┼───────────────────────────────────┤
; │ DSI           (display DSI)     │    7     │ SAFE       │ stm32h7xx_hal_msp.c               │
; │ LTDC          (display LTDC)    │    7     │ SAFE       │ stm32h7xx_hal_msp.c               │
; │ DMA2D         (display blend)   │    7     │ SAFE       │ stm32h7xx_hal_msp.c               │
; │ UART8         (BLE / NORA)      │    7     │ SAFE       │ stm32h7xx_hal_msp.c               │
; ├─────────────────────────────────┼──────────┼────────────┼───────────────────────────────────┤
; │ PendSV        (FreeRTOS switch) │   15     │ KERNEL     │ stm32h7xx_hal_msp.c               │
; └─────────────────────────────────┴──────────┴────────────┴───────────────────────────────────┘
;
; NOTE: JPEG was at 5 (tied with MDMA on the fence). Fixed to 6 so MDMA has priority
;       during large frame-buffer moves. Audio DMA stays at 5 = highest RTOS-safe slot.

; ══════════════════════════════════════════════════════════════════════════════
; Rule: higher number = higher priority (FreeRTOS convention, opposite of NVIC)
