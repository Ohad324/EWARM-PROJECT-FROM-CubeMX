; ─── MX_DFSDM1_Init (CM7/Core/Src/main.c ~line 502) ─────────────────────────

; DFSDM global enable (written manually, before Channel 3 init):
;   DFSDM1_Channel0->CHCFGR1 |= DFSDM_CHCFGR1_DFSDMEN;   // bit 31
;   DFSDM1_Channel0->CHCFGR1 |= (24u << DFSDM_CHCFGR1_CKOUTDIV_Pos);  // bits[23:16]
;   Effect: CKOUT = APB2 / ((24+1)×2) = 100 MHz / 50 = 2.0 MHz

; DFSDM1 Channel 3 settings:
;   SpiClock   = DFSDM_CHANNEL_SPI_CLOCK_INTERNAL   → uses CKOUT (2.0 MHz) as bit-clock
;   Multiplexer= DFSDM_CHANNEL_EXTERNAL_INPUTS       → data comes from SAI4 bridge on CH3
;   SerialType = DFSDM_CHANNEL_SPI_FALLING           → latch on falling edge (LR=GND)
;   Oversampling= 125                                → PCM = 2.0 MHz / 125 = 16,000 Hz
;   RightBitShift= 5                                 → Sinc3 peak 125³ ~21-bit → 16-bit

; DFSDM1 Filter 0 settings:
;   Trigger    = DFSDM_FILTER_SW_TRIGGER
;   FastMode   = ENABLE
;   DmaMode    = ENABLE
;   SincOrder  = DFSDM_FILTER_SINC3_ORDER
;   Oversampling= 125
;   Channel    = DFSDM_CHANNEL_3 (connected to SAI4 D1 bridge)

; ─── MX_SAI4_Init (CM7/Core/Src/main.c ~line 912) ───────────────────────────

;   AudioMode          = SAI_MODEMASTER_RX          → master receive (drives CK1, receives D1)
;   AudioFrequency     = SAI_AUDIO_FREQUENCY_16K    → HAL computes MCKDIV=12
;   PdmInit.Activation = ENABLE
;   PdmInit.MicPairs   = 1
;   PdmInit.ClockEnable= SAI_PDM_CLOCK1_ENABLE      → CK1 on PE2 = 2.048 MHz
;   SlotNumber         = 1
;   SlotActive         = SAI_SLOTACTIVE_0            → slot 0 active — SLOTR=0x00010000
;   FrameLength        = 8
;   ClockStrobing      = SAI_CLOCKSTROBING_FALLINGEDGE

; MCKDIV calculation:
;   PLL2P = 49.142857 MHz (HSE/M=25/25=1 MHz × N=344 / P=7)
;   MCKDIV = floor(PLL2P / (AudioFreq × 256 × MicPairs))
;          = floor(49142857 / (16000 × 256 × 1))
;          = floor(49142857 / 4096000)
;          = floor(12.0) = 12
;   CK1 = PLL2P / (MCKDIV × 2) = 49142857 / 24 = 2.047619 MHz ≈ 2.048 MHz ✓

; PCM rate check:
;   PDM clock = SAI4 CK1 = 2.048 MHz
;   DFSDM CKOUT = 2.0 MHz (±2.4% difference — acceptable)
;   PCM = CKOUT / OSR = 2.0 MHz / 125 = 16,000 Hz ✓
