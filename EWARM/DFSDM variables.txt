; ─── MX_DFSDM1_Init (main.c ~line 502) ──────────────────────────────────────
;
;   CORRECT WRITE ORDER (RM0399 §30.4.2):
;     CKOUTDIV must be written while DFSDMEN=0 — hardware silently drops
;     the write if DFSDMEN=1 is already set. (This was Bug 4 — now fixed.)
;
;   Code sequence (order is mandatory):
;     __HAL_RCC_DFSDM1_CLK_ENABLE();
;     DFSDM1_Channel0->CHCFGR1 |= (24u << DFSDM_CHCFGR1_CKOUTDIV_Pos);  // STEP 1
;     DFSDM1_Channel0->CHCFGR1 |= DFSDM_CHCFGR1_DFSDMEN;                 // STEP 2
;     HAL_DFSDM_ChannelInit(&hdfsdm1_channel3);                           // STEP 3
;
;   CKOUT = APB2 / ((CKOUTDIV+1) × 2) = 100 MHz / ((24+1)×2) = 2.0 MHz

; ─── DFSDM1 Channel 3 settings ───────────────────────────────────────────────
;   SpiClock    = DFSDM_CHANNEL_SPI_CLOCK_INTERNAL   → uses DFSDM CKOUT (2.0 MHz)
;   Multiplexer = DFSDM_CHANNEL_EXTERNAL_INPUTS       → data via SAI4 internal bridge
;   SerialType  = DFSDM_CHANNEL_SPI_FALLING           → latch on falling edge (LR=GND)
;   RightBitShift = 5                                 → Sinc3 125³ ~21-bit → 16-bit

; ─── DFSDM1 Filter 0 settings ────────────────────────────────────────────────
;   Trigger     = DFSDM_FILTER_SW_TRIGGER
;   FastMode    = ENABLE
;   DmaMode     = ENABLE
;   SincOrder   = DFSDM_FILTER_SINC3_ORDER
;   Oversampling = 125                                → PCM = 2.0 MHz / 125 = 16,000 Hz
;   Channel     = DFSDM_CHANNEL_3

; ─── MX_SAI4_Init (main.c ~line 912) ────────────────────────────────────────
;   AudioMode          = SAI_MODEMASTER_RX
;   AudioFrequency     = SAI_AUDIO_FREQUENCY_16K    → HAL computes MCKDIV=11
;   PdmInit.ClockEnable= SAI_PDM_CLOCK1_ENABLE      → CK1 on PE2
;   SlotActive         = SAI_SLOTACTIVE_0            → SLOTR=0x00010000
;   FrameLength        = 8
;
;   POST-INIT CR1 OVERRIDE (USER CODE BEGIN SAI4_Init 2):
;     hsai_BlockA4.Instance->CR1 = (CR1 & ~SAI_xCR1_MCKDIV) | (12u << SAI_xCR1_MCKDIV_Pos);
;     → Forces MCKDIV=12 because HAL rounds floor(49142857/4096000)=11 (not 12)
;     → CK1 = PLL2P / (2 × MCKDIV) = 49142857 / 24 = 2.048 MHz ✓

; ─── Clock math ──────────────────────────────────────────────────────────────
;   PLL2P = HSE/M × N / P = (25/25) × 344 / 7 = 49,142,857 Hz
;
;   HAL MCKDIV formula: floor(PLL2P / (AudioFreq × 256 × MicPairs))
;
;   | AudioFrequency | MCKDIV | CK1 (PE2)  | vs CKOUT=2.0 MHz | Status     |
;   |----------------|--------|------------|------------------|------------|
;   | 192000 (orig)  |   1    | 24.57 MHz  | +1129%           | BROKEN     |
;   | 250000         |   0    | undefined  | —                | BROKEN     |
;   | 256000 (Gemini)|   0    | undefined  | —                | BROKEN     |
;   | 16000 (16K)    |  11    |  2.234 MHz | +11.7%           | works/noisy|
;   | CR1 override   |  12    |  2.048 MHz | +2.4%            | TARGET ✓   |
;
;   DFSDM CKOUT = 100 MHz / ((24+1)×2) = 2.000 MHz
;   SAI4 CK1   = 49142857 / (2×12)    = 2.048 MHz  (+2.4% — acceptable for voice)
;   PCM rate   = CKOUT / OSR = 2.0 MHz / 125       = 16,000 Hz ✓

; 8. ALL BUGS FIXED — chronological
; ══════════════════════════════════════════════════════════════════════════════

; [Bug 1] DFSDM CKOUT disabled
;   SPICKSEL=INTERNAL but OutputClock.Activation=DISABLE → CKOUTDIV=0 → no clock
;   Fix: write CKOUTDIV=24 directly to Channel 0 CHCFGR1

; [Bug 2] SAI4 SlotActive = 0
;   No SAI4 slots → SAI4 generated CK1 but never captured D1 from PC1
;   → DFSDM internal bridge received nothing
;   Fix: hsai_BlockA4.SlotInit.SlotActive = SAI_SLOTACTIVE_0

; [Bug 3] SAI4 AudioFrequency wrong → MCKDIV=0 or MCKDIV=1
;   192K → MCKDIV=1 → CK1=24.6 MHz (way too fast)
;   250K / 256K → MCKDIV=0 → undefined
;   Fix: SAI_AUDIO_FREQUENCY_16K → MCKDIV=11 (valid, but 11.7% clock mismatch)

; [Bug 4] CKOUTDIV write silently dropped — RM0399 ordering violation
;   Code set DFSDMEN=1 first, then wrote CKOUTDIV
;   RM0399 §30.4.2: CKOUTDIV writable ONLY when DFSDMEN=0
;   Result: Ch0_CHCFGR1=0x80000000 (CKOUTDIV stayed 0 → CKOUT disabled → filter dead)
;   Fix: write CKOUTDIV first, then DFSDMEN=1

; [Bug 5] HAL MCKDIV rounding: floor(49142857/4096000) = 11, not 12
;   MCKDIV=11 → CK1=2.234 MHz → 11.7% faster than DFSDM CKOUT=2.0 MHz
;   Fix: force MCKDIV=12 via direct CR1 write after HAL_SAI_Init
;   MCKDIV=12 → CK1=2.048 MHz → only 2.4% from CKOUT (acceptable for voice)
